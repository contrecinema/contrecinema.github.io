const { createClient } = window.supabase; const db = createClient( window.SUPABASE_URL, window.SUPABASE_PUBLISHABLE_KEY ); const $ = id => document.getElementById(id); let issues = []; let sections = []; let articles = []; let team = []; let settings = null; async function login() { const { error } = await db.auth.signInWithPassword({ email: $("email").value, password: $("password").value }); if (error) { $("msg").textContent = error.message; } else { start(); } } async function logout() { await db.auth.signOut(); location.reload(); } async function start() { $("login").hidden = true; $("app").hidden = false; await loadAll(); tab("dash"); } db.auth.getSession() .then(({data}) => { if (data.session) start(); }); function tab(id) { document .querySelectorAll(".panel") .forEach(x => x.hidden = true ); $(id).hidden = false; } async function loadAll() { const result = await Promise.all([ db .from("issues") .select("*") .order( "number", {ascending:false} ), db .from("sections") .select("*") .order("sort_order"), db .from("articles") .select( "*,sections(name),issues(number)" ) .order( "created_at", {ascending:false} ), db .from("team") .select("*") .order("sort_order"), db .from("settings") .select("*") .eq("id",true) .maybeSingle() ]); issues = result[0].data || []; sections = result[1].data || []; articles = result[2].data || []; team = result[3].data || []; settings = result[4].data; render(); } function render() { $("stats").innerHTML = [ ["الأعداد",issues.length], ["الأقسام",sections.length], ["المقالات",articles.length], ["الهيئة",team.length] ].map(x => `
${x[0]}

${x[1]}

`).join(""); $("issuesList").innerHTML = issues.map(x => `
العدد ${x.number} — ${x.subtitle || ""}
${x.date_label || ""} · ${ x.pdf_url ? "PDF مرفوع" : "PDF غير مرفوع" }
`).join(""); $("sectionsList").innerHTML = sections.map(x => `
${x.name}
${x.description || ""}
`).join(""); $("articlesList").innerHTML = articles.map(x => `
${x.title} — ${x.author || ""}
`).join(""); $("teamList").innerHTML = team.map(x => `
${x.name} — ${x.role || ""}
`).join(""); $("articleSection").innerHTML = '' + sections.map(x => ` `).join(""); $("articleIssue").innerHTML = '' + issues.map(x => ` `).join(""); $("currentIssue").innerHTML = issues.map(x => ` `).join(""); if (settings) fillSettings(); } async function upload(bucket,file) { if (!file) return null; const path = Date.now() + "-" + file.name.replace( /[^a-zA-Z0-9._-]/g, "-" ); const {error} = await db.storage .from(bucket) .upload( path, file ); if (error) throw error; return db.storage .from(bucket) .getPublicUrl(path) .data.publicUrl; } async function saveIssue(e) { e.preventDefault(); try { const id = $("issueId").value; const old = issues.find( x => x.id === id ); let cover = old?.cover_url || null; let pdf = old?.pdf_url || null; let pdfName = old?.pdf_name || null; if ($("coverFile").files[0]) cover = await upload( "covers", $("coverFile").files[0] ); if ($("pdfFile").files[0]) { pdf = await upload( "pdfs", $("pdfFile").files[0] ); pdfName = $("pdfFile").files[0].name; } const row = { number: Number( $("issueNumber").value ), date_label: $("issueDate").value, subtitle: $("issueSubtitle").value, description: $("issueDescription").value, cover_url: cover, pdf_url: pdf, pdf_name: pdfName, published: $("issuePublished").checked }; const result = id ? await db .from("issues") .update(row) .eq("id",id) : await db .from("issues") .insert(row); if (result.error) throw result.error; resetIssue(); await loadAll(); } catch(error) { alert(error.message); } } function editIssue(id) { const x = issues.find( a => a.id === id ); $("issueId").value = x.id; $("issueNumber").value = x.number; $("issueDate").value = x.date_label || ""; $("issueSubtitle").value = x.subtitle || ""; $("issueDescription").value = x.description || ""; $("issuePublished").checked = x.published; } function resetIssue() { $("issueId").value = ""; [ "issueNumber", "issueDate", "issueSubtitle", "issueDescription" ].forEach( x => $(x).value = "" ); $("coverFile").value = ""; $("pdfFile").value = ""; $("issuePublished").checked = true; } async function saveSection(e) { e.preventDefault(); const id = $("sectionId").value; const row = { name: $("sectionName").value, description: $("sectionDesc").value }; const result = id ? await db .from("sections") .update(row) .eq("id",id) : await db .from("sections") .insert(row); if (result.error) alert(result.error.message); else { $("sectionId").value = ""; $("sectionName").value = ""; $("sectionDesc").value = ""; loadAll(); } } function editSection(id) { const x = sections.find( a => a.id === id ); $("sectionId").value = x.id; $("sectionName").value = x.name; $("sectionDesc").value = x.description || ""; } async function saveArticle(e) { e.preventDefault(); const id = $("articleId").value; const row = { title: $("articleTitle").value, author: $("articleAuthor").value, excerpt: $("articleExcerpt").value, body: $("articleBody").value, section_id: $("articleSection").value || null, issue_id: $("articleIssue").value || null, published: $("articlePublished").checked }; const result = id ? await db .from("articles") .update(row) .eq("id",id) : await db .from("articles") .insert(row); if (result.error) alert(result.error.message); else { resetArticle(); loadAll(); } } function editArticle(id) { const x = articles.find( a => a.id === id ); $("articleId").value = x.id; $("articleTitle").value = x.title; $("articleAuthor").value = x.author || ""; $("articleExcerpt").value = x.excerpt || ""; $("articleBody").value = x.body || ""; $("articleSection").value = x.section_id || ""; $("articleIssue").value = x.issue_id || ""; $("articlePublished").checked = x.published; } function resetArticle() { $("articleId").value = ""; [ "articleTitle", "articleAuthor", "articleExcerpt", "articleBody" ].forEach( x => $(x).value = "" ); $("articlePublished").checked = true; } async function saveTeam(e) { e.preventDefault(); const id = $("teamId").value; const row = { name: $("teamName").value, role: $("teamRole").value, bio: $("teamBio").value, sort_order: Number( $("teamOrder").value ) || 0 }; const result = id ? await db .from("team") .update(row) .eq("id",id) : await db .from("team") .insert(row); if (result.error) alert(result.error.message); else loadAll(); } function editTeam(id) { const x = team.find( a => a.id === id ); $("teamId").value = x.id; $("teamName").value = x.name; $("teamRole").value = x.role || ""; $("teamBio").value = x.bio || ""; $("teamOrder").value = x.sort_order || 0; } async function del(table,id) { if ( !confirm( "هل تريد الحذف؟" ) ) return; const result = await db .from(table) .delete() .eq("id",id); if (result.error) alert( result.error.message ); else loadAll(); } function fillSettings() { const s = settings; $("siteName").value = s.site_name || ""; $("heroTitleInput").value = s.hero_title || ""; $("heroSubtitleInput").value = s.hero_subtitle || ""; $("heroDescriptionInput").value = s.hero_description || ""; $("aboutTitleInput").value = s.about_title || ""; $("aboutTextInput").value = s.about_text || ""; $("aboutText2Input").value = s.about_text_2 || ""; $("contactEmailInput").value = s.contact_email || ""; $("currentIssue").value = s.current_issue_id || ""; } async function saveSettings(e) { e.preventDefault(); try { let hero = settings?.hero_cover_url || null; if ( $("heroFile").files[0] ) { hero = await upload( "covers", $("heroFile").files[0] ); } const row = { id: true, site_name: $("siteName").value, hero_title: $("heroTitleInput").value, hero_subtitle: $("heroSubtitleInput").value, hero_description: $("heroDescriptionInput").value, about_title: $("aboutTitleInput").value, about_text: $("aboutTextInput").value, about_text_2: $("aboutText2Input").value, contact_email: $("contactEmailInput").value, current_issue_id: $("currentIssue").value || null, hero_cover_url: hero }; const result = await db .from("settings") .upsert(row); if (result.error) throw result.error; await loadAll(); alert("تم الحفظ"); } catch(error) { alert(error.message); } }
